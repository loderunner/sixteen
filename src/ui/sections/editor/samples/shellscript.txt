#!/usr/bin/env bash
set -euo pipefail

# Install the theme in every theme folder we find.
readonly THEME="harbor"
readonly SOURCE="${1:-./themes}"

install_theme() {
  local target="$1"
  mkdir -p "$target" && cp "$SOURCE/$THEME"-*.json "$target/"
  echo "Installed $THEME in $target"
}

for folder in "$HOME/.config/themes" "$HOME/.local/share/themes"; do
  if [[ -d "$folder" ]]; then
    install_theme "$folder/$THEME"
  fi
done

count=$(find "$SOURCE" -name '*.json' | wc -l)
printf 'Done: %d files\n' "$count"
exit 0
